```python
"""LLM-powered structured data extraction using JSON schema."""

import json
from loguru import logger

_MAX_CONTENT_CHARS = 50_000


async def extract_structured(
    urls: list[str],
    schema: dict,
    prompt: str | None = None,
    stealth: bool = False,
) -> str:
    """Extract structured data from URLs using LLM + JSON schema.

    Pipeline:
    1. Extract raw content (markdown) from URLs
    2. Send content + schema to LLM with response_format=json_schema
    3. Validate output with jsonschema
    4. Return JSON result

    Requires LLM (proxy or SDK mode). Returns error in local mode.
    """
    from wet_mcp.config import settings

    mode = settings.resolve_litellm_mode()
    if mode == "local":
        return json.dumps({
            "error": "Structured extraction requires LLM. Configure LITELLM_PROXY_URL or API_KEYS."
        })

    # Step 1: extract raw content
    from wet_mcp.sources.crawler import extract as raw_extract
    raw_json = await raw_extract(urls=urls, format="markdown", stealth=stealth)
    pages = json.loads(raw_json)

    # Combine content
    contents = []
    for page in pages:
        if isinstance(page, dict) and "error" not in page:
            content = page.get("content", page.get("markdown", ""))
            title = page.get("title", "")
            url = page.get("url", "")
            contents.append(f"## {title}\nURL: {url}\n\n{content}")

    if not contents:
        return json.dumps({"error": "No content could be extracted from provided URLs"})

    combined = "\n\n---\n\n".join(contents)
    if len(combined) > _MAX_CONTENT_CHARS:
        combined = combined[:_MAX_CONTENT_CHARS] + "\n...[truncated]"

    # Step 2: call LLM
    result_data = await _call_llm_with_schema(combined, schema, prompt)
    if result_data is None:
        return json.dumps({"error": "LLM extraction failed"})

    # Step 3: validate
    import jsonschema as _jsonschema
    try:
        _jsonschema.validate(result_data, schema)
    except _jsonschema.ValidationError as ve:
        return json.dumps({
            "data": result_data,
            "validation_warning": ve.message,
            "urls": urls,
        }, ensure_ascii=False, indent=2)

    return json.dumps({
        "data": result_data,
        "urls": urls,
    }, ensure_ascii=False, indent=2)


async def _call_llm_with_schema(
    content: str, schema: dict, prompt: str | None
) -> dict | None:
    """Call LLM with json_schema response format, fallback to json_object."""
    from litellm import acompletion
    from wet_mcp.llm import get_llm_config

    config = get_llm_config()
    system_prompt = (
        "You are a structured data extraction assistant. "
        "Extract information from the provided web content according to the given schema. "
        "Return ONLY valid JSON matching the schema. "
        "If a field cannot be found, use null."
    )
    user_prompt = "Extract structured data from the following web content.\n\n"
    if prompt:
        user_prompt += f"Additional instructions: {prompt}\n\n"
    user_prompt += f"Content:\n{content}"

    schema_name = schema.get("title", "extraction_result")

    # Try json_schema mode first
    try:
        response = await acompletion(
            model=config["model"],
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt},
            ],
            response_format={
                "type": "json_schema",
                "json_schema": {
                    "name": schema_name,
                    "schema": schema,
                    "strict": False,
                },
            },
            temperature=0,
            **{k: v for k, v in config.items() if k not in ("model", "fallbacks", "temperature")},
        )
        return json.loads(response.choices[0].message.content)
    except Exception as e:
        logger.debug(f"json_schema mode failed, trying json_object fallback: {e}")

    # Fallback: json_object mode with schema in prompt
    try:
        schema_str = json.dumps(schema, indent=2)
        fallback_prompt = (
            f"{user_prompt}\n\n"
            f"Return JSON matching this schema:\n```json\n{schema_str}\n```"
        )
        response = await acompletion(
            model=config["model"],
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": fallback_prompt},
            ],
            response_format={"type": "json_object"},
            temperature=0,
            **{k: v for k, v in config.items() if k not in ("model", "fallbacks", "temperature")},
        )
        return json.loads(response.choices[0].message.content)
    except Exception as e:
        logger.error(f"Structured extraction failed: {e}")
        return None
```
